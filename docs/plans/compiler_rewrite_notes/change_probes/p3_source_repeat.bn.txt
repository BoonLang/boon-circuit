            input |> THEN { count + 1 }

        0 |> HOLD payload_count {
            input.text |> THEN { payload_count + 1 }
        }

        }

        }

        LATEST {
            TEXT { init }
            input |> THEN { input.text }
        }

        0 |> HOLD latest_fires {
            latest_text |> THEN { latest_fires + 1 }
        }
]
