store: [
    sources: [
        a_button: [events: [press: SOURCE]]
        b_button: [events: [press: SOURCE]]
    ]
]

tallies: [
    a: tally(press: store.sources.a_button.events.press)
    b: tally(press: store.sources.b_button.events.press)
]

FUNCTION tally(press) {
    [
        count:
            0 |> HOLD n {
                press |> THEN { n + 1 }
            }
    ]
}

document: Document/new(
    root: Element/stripe(
        element: []
        direction: Column
        gap: 4
        style: []
        items: LIST {
            Element/label(element: [], style: [], label: tallies.a.count)
            Element/label(element: [], style: [], label: tallies.b.count)
        }
    )
)
