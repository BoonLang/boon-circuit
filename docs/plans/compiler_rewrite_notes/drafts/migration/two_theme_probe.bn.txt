store: [
    toggle: SOURCE
    pick: SOURCE
    mode: Light |> HOLD state {
        toggle |> THEN { state |> WHEN { Light => Dark, Dark => Light } }
    }
    name: Classic |> HOLD name {
        pick |> THEN { Professional }
    }
]

theme: store.name |> WHEN {
    Classic => classic_tokens(mode: store.mode)
    Professional => prof_tokens(mode: store.mode)
}

document: Document/new(root: panel())

FUNCTION panel() {
    Element/stripe(
        element: []
        direction: Column
        gap: theme.spacing.tight
        style: [
            width: 300
            depth: theme.depth.container
            rounded_corners: theme.corners.pill
            material: theme.material.panel
            shadows: theme.panel_shadows
            ...theme.frame.footer
        ]
        items: LIST {
            Element/label(element: [], style: [font: theme.font.body, color: theme.font.body_disabled.color, ...theme.frame.todo_row], label: TEXT { row })
            Element/checkbox(element: [], style: [size: 40, ...theme.checkbox], label: TEXT { c }, checked: False)
            Element/label(element: [], style: [font: [...theme.font.small, line: [underline: False]], material: theme.material.editing, outline: [color: theme.primary_color]], label: TEXT { hi })
        }
    )
}

-- Neutral records shared by every theme. Each value equals what the renderer
-- uses when the field is missing, so `[...Base/material(), color: X]` renders
-- exactly like `[color: X]` while every material keeps one shape.

FUNCTION base_material() {
    [
        color: None
        gloss: 0
        metal: 0
        refraction: 0
        transparency: 0
        frosted_blur: 0
        frosted_saturate: 1
        glass_highlight: 0
        glass_highlight_color: TEXT { #ffffff }
    ]
}

-- Edges spread into a row or footer style. None draws nothing; every themed
-- side border sets its own width, so the zero widths are never read.
FUNCTION base_frame() {
    [
        border: None
        border_width: 0
        border_top: None
        border_top_width: 0
        border_bottom: None
        border_bottom_width: 0
        shadows: LIST {}
    ]
}

-- Classic Theme
-- Flat scene materials shaped to match the document-api TodoMVC reference.

FUNCTION classic_tokens(mode) {
    BLOCK {
        shadow_medium: mode |> WHEN {
            Light => Oklch[lightness: 0, chroma: 0, hue: 0, alpha: 0.24]
            Dark => Oklch[lightness: 0, chroma: 0, hue: 0, alpha: 0.45]
        }
        shadow_soft: mode |> WHEN {
            Light => Oklch[lightness: 0, chroma: 0, hue: 0, alpha: 0.14]
            Dark => Oklch[lightness: 0, chroma: 0, hue: 0, alpha: 0.30]
        }
        primary: mode |> WHEN {
            Light => Oklch[lightness: 0.585, chroma: 0.172, hue: 24.1]
            Dark => TEXT { #b83f45 }
        }

        [
            material: classic_materials(mode: mode, primary: primary)
            panel_shadows: LIST {
                [x: 0, y: 2, blur: 4, spread: 0, color: shadow_medium]
                [x: 0, y: 25, blur: 50, spread: 0, color: shadow_soft]
            }
            frame: classic_frames(mode: mode, shadow_medium: shadow_medium)
            checkbox: classic_checkbox(mode: mode)
            font: classic_fonts(mode: mode)
            primary_color: primary
            depth: [container: 0, element: 0, detail: 0, hero: 0]
            elevation: [card: 0, editing_focus: 0, selection: 0, todo_item: 0]
            corners: [touch: 0, comfort: 0, filter: 3, pill: 3]
            sizing: [toggle_control: 45]
            spacing: [tight: 11, standard: 9, new_todo_input_inset: 15]
            spring: [
                button: [extend: 0, compress: 0]
                button_destructive: [extend: 0, compress: 0]
                checkbox: [extend: 0, compress: 0]
            ]
            geometry: [edge_radius: 0, bevel_angle: 0]
        ]
    }
}

FUNCTION classic_materials(mode, primary) {
    BLOCK {
        surface_color: mode |> WHEN { Light => Oklch[lightness: 1], Dark => TEXT { #202020 } }
        surface_base: [...base_material(), color: surface_color, gloss: 0]

        [
            background: [
                ...base_material()
                color: mode |> WHEN { Light => Oklch[lightness: 0.97], Dark => TEXT { #1b1b1b } }
            ]
            surface: surface_base
            surface_variant: surface_base
            surface_elevated: surface_base
            panel: [...surface_base, border: None, border_width: 1]
            interactive: surface_base
            interactive_hovered: surface_base

            filter_selected: [...base_material(), border: primary, border_width: 1]
            hero_title: [...base_material(), color: TEXT { #00000000 }]

    }
}

-- Professional Theme
-- Soft rounded edges, subtle shadows, low gloss materials
-- Neutral warm lighting, classic design system aesthetic

FUNCTION prof_tokens(mode) {
    [
        material: prof_materials(mode: mode)
        panel_shadows: mode |> WHEN {
            Light => LIST {
                [x: 0, y: 24, blur: 48, spread: 0, color: TEXT { #222e4424 }]
                [x: 0, y: 2, blur: 7, spread: 0, color: TEXT { #222e4414 }]

            Dark => LIST {
                [x: 0, y: 26, blur: 52, spread: 0, color: TEXT { #0000006b }]
                [x: 0, y: 1, blur: 0, spread: 0, color: TEXT { #ffffff08 }]
            }
        }
        frame: prof_frames(mode: mode)
        checkbox: prof_checkbox(mode: mode)
        font: prof_fonts(mode: mode)
        primary_color: mode |> WHEN { Light => TEXT { #bd454d }, Dark => TEXT { #dc6463 } }
        depth: [container: 8, element: 6, detail: 2, hero: 10]
        elevation: [card: 50, editing_focus: 24, selection: 4, todo_item: 4]
        corners: [
            touch: 8
            comfort: 8
            filter: mode |> WHEN { Light => 999, Dark => 6 }
            pill: Fully
